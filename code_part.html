38    @Override
39    public Map<String, String> labelParts(Integer modelId, PartLabelRequest request) {
40        BuildingModel model = modelRepository.findById(modelId)
41            .orElseThrow(() -> new IllegalArgumentException("模型不存在"));
42
43        // parts 是前端传来的 glTF 节点/网格列表：id 是 three.js 侧的节点 id，technicalName 是 glTF 导出的技术名
44        List<PartLabelRequest.PartItem> parts = request.getParts();
45        String lines = parts.stream()
46            .map(p -> p.getId() + "\t" + p.getTechnicalName())
47            .collect(Collectors.joining("\n"));
48
49        // 强制只输出 JSON，方便直接反序列化成 Map
50        String system = "你是中国古建筑领域的构件命名助手。用户会提供 glTF 导出常见的英文或混合命名的网格/节点名称。"
51            + "请结合名称给出简短、专业的中文构件称呼（2~10个汉字为宜）。"
52            + "若名称过于模糊，给出中性但合理的称呼（如「木构件」「石构件」），不要编造具体历史信息。"
53            + "输出要求：只输出一个 JSON 对象，不要 markdown 代码块，不要解释。"
54            + "JSON 的键必须是用户给出的 id（原样字符串），值是对应的中文名称。";
55
56        StringBuilder user = new StringBuilder();
57        user.append("建筑模型：").append(model.getModelName() != null ? model.getModelName() : "未命名").append('\n');
58        if (model.getDynasty() != null) {
59            user.append("朝代：").append(model.getDynasty()).append('\n');
60        }
61        if (model.getLocation() != null) {
62            user.append("位置：").append(model.getLocation()).append('\n');
63        }
64        user.append("部件列表（每行：id<TAB>技术名）：\n");
65        user.append(lines);
66
67        String raw;
68        try {
69            // 非流式调用，一次拿到完整的 JSON
70            raw = deepSeekClient.completeChat(system, user.toString());
71        } catch (Exception e) {
72            log.warn("Model part label DeepSeek failed: modelId={}, msg={}", modelId, e.getMessage());
73            throw new IllegalStateException("AI 命名服务暂时不可用: " + e.getMessage());
74        }
75
76        // AI 偶尔会在 JSON 外面裹点说明文字，把最外层的 {...} 抠出来再解析
77        String jsonBlob = extractJsonObject(raw.trim());
78        JsonNode node;
79        try {
80            node = objectMapper.readTree(jsonBlob);
81        } catch (Exception e) {
82            log.warn("Model part label JSON parse failed: modelId={}, rawPrefix={}",
83                modelId, jsonBlob.length() > 200 ? jsonBlob.substring(0, 200) : jsonBlob);
84            throw new IllegalStateException("AI 返回内容无法解析为 JSON，请稍后重试");
85        }
86
87        Map<String, String> out = new LinkedHashMap<>();
88        for (PartLabelRequest.PartItem p : parts) {
89            JsonNode v = node.get(p.getId());
90            if (v != null && v.isTextual()) {
91                String label = v.asText().trim();
92                if (!label.isEmpty()) {
93                    out.put(p.getId(), label);
94                    continue;
95                }
96            }
97            // AI 漏了某个 id 或者给了空串时退回技术名，前端至少有东西可显示
98            out.put(p.getId(), p.getTechnicalName());
99        }
100        return out;
101    }

161    private List<PartBlurbSourceItem> buildPartBlurbSources(
162            BuildingModel model,
163            String technicalName,
164            String aiLabel) {
165
166        List<PartBlurbSourceItem> sources = new ArrayList<>();
167
168        String modelTitle = model.getModelName() != null ? model.getModelName() : "未命名";
169        StringBuilder metaLine = new StringBuilder(modelTitle);
170        if (model.getDynasty() != null && !model.getDynasty().isBlank()) {
171            metaLine.append(" · ").append(model.getDynasty().trim());
172        }
173        if (model.getLocation() != null && !model.getLocation().isBlank()) {
174            metaLine.append(" · ").append(model.getLocation().trim());
175        }
176        sources.add(PartBlurbSourceItem.builder()
177            .label("模型条目")
178            .detail(metaLine.toString())
179            .build());
180
181        if (model.getDescription() != null && !model.getDescription().isBlank()) {
182            String sd = model.getDescription().trim();
183            String excerpt = sd.length() > 140 ? sd.substring(0, 140) + "…" : sd;
184            sources.add(PartBlurbSourceItem.builder()
185                .label("模型简介（节选）")
186                .detail(excerpt)
187                .build());
188        }
189
190        sources.add(PartBlurbSourceItem.builder()
191            .label("构件节点（glTF）")
192            .detail(technicalName != null ? technicalName : "")
193            .build());
194
195        if (aiLabel != null && !aiLabel.isBlank()) {
196            sources.add(PartBlurbSourceItem.builder()
197                .label("中文称谓（AI）")
198                .detail(aiLabel.trim())
199                .build());
200        }
201
202        sources.add(PartBlurbSourceItem.builder()
203            .label("摘要引擎")
204            .detail("DeepSeek Chat · 依据上述字段推断生成")
205            .build());
206
207        return sources;
208    }

210    // AI 输出不保证干净，掐头去尾只留最外层 JSON 对象
211    private String extractJsonObject(String raw) {
212        int start = raw.indexOf('{');
213        int end = raw.lastIndexOf('}');
214        if (start < 0 || end <= start) {
215            throw new IllegalStateException("未找到 JSON 对象");
216        }
217        return raw.substring(start, end + 1);
218    }
