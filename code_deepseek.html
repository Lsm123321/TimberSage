71     * 返回的是 SSE 风格的行（"data: " 开头，最后一条是 "[DONE]"），

76     * 回调有"只允许完整走一轮"的幂等保护：备胎开讲后主线程残留的读循环不会再推内容。
77     */
78    public void streamChat(String prompt, String modelId, String modelName, String dynasty, String location,
79                          DeepSeekCallback callback) {
80        Instant start = Instant.now();
81        AtomicBoolean delivered = new AtomicBoolean(false);
82        try {
83            // 主备策略：DeepSeek 共享通道高峰期会 503，或首 token 后慢到 1 字/20 秒（8 分钟吐不完一段回答，
84            // SSE 连接 5 分钟就超时）。智谱 glm-4-flash 实测稳定，所以平时主用智谱，
85            // 智谱失败再回退 DeepSeek——两个通道都走同样的 OpenAI 兼容协议。
86            String zKey = Objects.toString(zhipuApiKey, "").trim();
87            Exception primaryError = null;
88            if (!zKey.isEmpty()) {
89                primaryError = tryStreamChat(zhipuChatUrl, zKey, zhipuChatModel,
90                        buildSystemPrompt(modelName, dynasty, location), prompt, callback, delivered);
91                if (primaryError == null) {
92                    return; // 智谱完整走完
93                }
94                log.warn("智谱流式调用失败，尝试 DeepSeek 通道: {}", primaryError.getMessage());
95            }
96
97            Exception backupError = tryStreamChat(apiUrl, Objects.toString(apiKey, "").trim(), "deepseek-chat",
98                    buildSystemPrompt(modelName, dynasty, location), prompt, callback, delivered);
99            if (backupError != null) {
100                callback.onError(primaryError != null ? primaryError : backupError);
101            }
102        } catch (Exception e) {
103            log.error("Error calling LLM API", e);
104            callback.onError(e);
105        }
106    }
