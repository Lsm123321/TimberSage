32    private static final String CACHE_KEY_PREFIX = "tour:script:";
33    /** 缓存 24 小时；模型信息更新后可由管理端清理或自然过期 */
34    private static final Duration CACHE_TTL = Duration.ofHours(24);

36    /**
37     * 生成（或命中缓存）导览脚本。
38     *
39     * @param modelName  模型名称
40     * @param dynasty    朝代
41     * @param location   地理位置
42     * @param description 模型描述
43     * @param partNames  前端上报的真实构件名清单（mesh 命名），供 LLM 规划高亮镜头
44     */
45    public TourScriptResponse getOrGenerateScript(String modelName, String dynasty,
46                                                  String location, String description,
47                                                  List<String> partNames) {
48        String cacheKey = CACHE_KEY_PREFIX + modelName;
49        try {
50            Object cached = redisTemplate.opsForValue().get(cacheKey);
51            if (cached instanceof TourScriptResponse) {
52                return (TourScriptResponse) cached;
53            }
54        } catch (Exception e) {
55            // Redis 不可用不影响主流程，直接走生成
56            log.warn("导览脚本缓存读取失败，将直接生成: {}", e.getMessage());
57        }
58
59        TourScriptResponse script = generateWithFallback(modelName, dynasty, location, description, partNames);
60        try {
61            redisTemplate.opsForValue().set(cacheKey, script, CACHE_TTL);
62        } catch (Exception e) {
63            log.warn("导览脚本缓存写入失败: {}", e.getMessage());
64        }
65        return script;
66    }

68    /** 先尝试 LLM 生成，解析失败再退回内置模板，保证接口总能返回可用脚本 */
69    private TourScriptResponse generateWithFallback(String modelName, String dynasty,
70                                                    String location, String description,
71                                                    List<String> partNames) {
72        try {
73            String content = deepSeekClient.completeChat(
74                buildSystemPrompt(),
75                buildUserPrompt(modelName, dynasty, location, description, partNames),
76                0.5, 3000);
77            TourScriptResponse parsed = parseScript(content, modelName);
78            if (parsed != null && parsed.getChapters() != null && !parsed.getChapters().isEmpty()) {
79                return parsed;
80            }
81            log.warn("LLM 返回的导览脚本解析为空，使用内置模板");
82        } catch (Exception e) {
83            log.error("生成导览脚本失败，使用内置模板", e);
84        }
85        return fallbackScript(modelName);
86    }

109    private String buildUserPrompt(String modelName, String dynasty, String location,
110                                   String description, List<String> partNames) {
111        StringBuilder sb = new StringBuilder();
112        sb.append("模型名称：").append(safe(modelName)).append("\n");
113        sb.append("朝代：").append(safe(dynasty)).append("\n");
114        sb.append("地理位置：").append(safe(location)).append("\n");
115        if (description != null && !description.isBlank()) {
116            String desc = description.length() > 400 ? description.substring(0, 400) + "…" : description;
117            sb.append("模型描述：").append(desc).append("\n");
118        }
119        if (partNames != null && !partNames.isEmpty()) {
120            // 名字太长会撑爆 prompt，截一下
121            String joined = String.join("、", partNames);
122            if (joined.length() > 600) joined = joined.substring(0, 600);
123            sb.append("构件名称清单（partName 只能从中选择）：").append(joined).append("\n");
124        }
125        sb.append("请输出导览脚本 JSON 数组。");
126        return sb.toString();
127    }

129    /**
130     * LLM 的输出不保证干净：可能裹着 ```json 围栏或者前后说明，
131     * 这里抠出最外层的 [...] 再反序列化，宽容一点少踩坑。
132     */
133    private TourScriptResponse parseScript(String content, String modelName) {
134        if (content == null || content.isBlank()) return null;
135        String text = content.trim();
136        // 剥掉 ```json ... ``` 围栏
137        Pattern fence = Pattern.compile("```(?:json)?\\s*([\\s\\S]*?)```", Pattern.CASE_INSENSITIVE);
138        Matcher m = fence.matcher(text);
139        if (m.find()) text = m.group(1).trim();
140        // 截取最外层数组
141        int start = text.indexOf('[');
142        int end = text.lastIndexOf(']');
143        if (start < 0 || end <= start) return null;
144        String arrayJson = text.substring(start, end + 1);
145        try {
146            JsonNode arr = objectMapper.readTree(arrayJson);
147            List<TourScriptResponse.TourChapter> chapters = new ArrayList<>();
148            for (JsonNode node : arr) {
149                String chapter = node.path("chapter").asText(null);
150                String narration = node.path("narration").asText(null);
151                if (chapter == null || narration == null) continue;
152                TourScriptResponse.TourChapter.TourChapterBuilder cb = TourScriptResponse.TourChapter.builder()
153                    .chapter(chapter)
154                    .narration(narration);
155                if (node.hasNonNull("partName")) cb.partName(node.get("partName").asText());
156                if (node.has("cameraPos") && node.get("cameraPos").isArray() && node.get("cameraPos").size() == 3) {
157                    cb.cameraPos(toDoubleList(node.get("cameraPos")));
158                }
159                if (node.has("cameraTarget") && node.get("cameraTarget").isArray() && node.get("cameraTarget").size() == 3) {
160                    cb.cameraTarget(toDoubleList(node.get("cameraTarget")));
161                }
162                if (node.hasNonNull("duration") && node.get("duration").isNumber()) {
163                    cb.duration(node.get("duration").asDouble(10.0));
164                }
165                if (node.hasNonNull("action")) cb.action(node.get("action").asText());
166                chapters.add(cb.build());
167            }
168            if (chapters.isEmpty()) return null;
169            return TourScriptResponse.builder()
170                .modelName(modelName)
171                .chapters(chapters)
172                .build();
173        } catch (Exception e) {
174            log.warn("导览脚本 JSON 解析失败: {}", e.getMessage());
175            return null;
176        }
177    }

179    private List<Double> toDoubleList(JsonNode arr) {
180        List<Double> list = new ArrayList<>(3);
181        for (JsonNode n : arr) list.add(n.asDouble(0.0));
182        return list;
183    }
